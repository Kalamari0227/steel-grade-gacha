var GAME_PWA = (function(){
  var registration, requested = false, available = false, explicitBusy = false, blocked = false;
  var previousController = navigator.serviceWorker && navigator.serviceWorker.controller;
  var notice = document.getElementById('gameUpdate'), message = document.getElementById('gameUpdateText');
  function busy(){
    return explicitBusy || (typeof saveHold !== 'undefined' && saveHold) ||
      (typeof drawBusy === 'function' && drawBusy()) ||
      (typeof dialogStack !== 'undefined' && dialogStack.length > 0);
  }
  function refresh(){
    if(registration && registration.waiting && navigator.serviceWorker.controller) available = true;
    notice.hidden = !available;
    if(available) message.textContent = blocked ? '다른 창의 진행 중인 작업을 마친 뒤 다시 눌러 주세요.' : busy() ? '진행 중인 작업을 마친 뒤 업데이트할 수 있습니다.' : '새 버전이 준비되었습니다. 원할 때 적용해 주세요.';
  }
  function state(){ return {type:'GAME_CLIENT_STATE',releaseId:GAME_CONFIG.releaseId,idle:!busy()}; }
  if('serviceWorker' in navigator){
    navigator.serviceWorker.addEventListener('message',function(event){
      if(event.data && event.data.type === 'GAME_QUERY_STATE' && event.ports[0]) event.ports[0].postMessage(state());
      if(event.data && event.data.type === 'GAME_UPDATE_BUSY'){
        requested = false; blocked = true; refresh();
      }
    });
    navigator.serviceWorker.addEventListener('controllerchange',function(){
      var hadController = !!previousController; previousController = navigator.serviceWorker.controller;
      if(requested && !busy()){ requested = false; location.reload(); }
      else if(hadController){ available = true; refresh(); }
    });
    window.addEventListener('load',function(){
      navigator.serviceWorker.register('sw.js',{updateViaCache:'none'}).then(function(reg){
        registration = reg; refresh();
        reg.addEventListener('updatefound',function(){
          var worker = reg.installing;
          if(worker) worker.addEventListener('statechange',refresh);
        });
      }).catch(function(){});
    });
    document.getElementById('gameUpdateButton').addEventListener('click',function(){
      blocked = false;
      if(busy()){ refresh(); return; }
      if(registration && registration.waiting){ requested = true; registration.waiting.postMessage({type:'GAME_ACTIVATE'}); }
      else if(available){ location.reload(); }
    });
    setInterval(refresh,500);
  }
  document.addEventListener('error',function(event){
    var img = event.target;
    if(!(img instanceof HTMLImageElement) || img.closest('.game-card-art,.craft-art,.garage-art,.home-figure,.home-car') || !/assets\/game\//.test(img.src) || img.dataset.gameFallback) return;
    img.dataset.gameFallback = 'true'; img.hidden = true;
    var fallback = document.createElement('span'), label = document.createElement('span'), retry = document.createElement('button');
    fallback.className = 'game-asset-fallback'; label.textContent = (img.alt ? img.alt + ' 그림을' : '그림을') + ' 불러오지 못했습니다. 인터넷 연결을 확인한 뒤 다시 시도해 주세요.';
    retry.type = 'button'; retry.textContent = '그림 다시 불러오기';
    retry.addEventListener('click',function(event){
      event.stopPropagation();
      var restoreFocus = fallback.contains(document.activeElement);
      retry.disabled = true;
      var url = img.src;
      img.addEventListener('load',function loaded(){
        img.hidden=false;delete img.dataset.gameFallback;
        if((fallback.contains(document.activeElement) || (restoreFocus && document.activeElement === document.body)) && host.matches('button,a')) host.focus();
        fallback.remove();
      },{once:true});
      img.addEventListener('error',function(){retry.disabled=false;},{once:true});
      img.src = url;
    });
    fallback.append(label,retry);
    var host = img.closest('button,a') || img;
    host.insertAdjacentElement('afterend',fallback);
  },true);
  return Object.freeze({isBusy:busy,setBusy:function(value){explicitBusy=!!value;refresh();}});
}());

<script>
(function(){

  var bar = document.getElementById("installBar");
  var go = document.getElementById("installGo");
  var x = document.getElementById("installX");
  var title = document.getElementById("installTitle");
  var sub = document.getElementById("installSub");
  var steps = document.getElementById("installSteps");
  var foot = document.getElementById("btnInstall");
function closeInstall(){
  if(dialogStack.some(function(entry){return entry.el===bar;})) hideDialog(bar);
  else bar.classList.remove("show");
  bar.setAttribute("role","region");
  bar.removeAttribute("aria-modal");
  steps.classList.remove("show");
}
dialogClosers.installBar = closeInstall;

  var deferred = null;
  var DISMISS = GAME_EDITION.installDismissed;
  var DONE = GAME_EDITION.installed;

  function flag(k){ try{ return localStorage.getItem(k)==="1"; }catch(e){ return false; } }
  function mark(k){ try{ localStorage.setItem(k,"1"); }catch(e){} }

  var standalone = window.matchMedia("(display-mode: standalone)").matches || navigator.standalone === true;
  var ua = navigator.userAgent || "";
  /* 아이패드 사파리는 맥인 척하므로 터치 여부로 가린다 */
  var isIOS = /iPad|iPhone|iPod/.test(ua) || (/Macintosh/.test(ua) && navigator.maxTouchPoints > 1);
  var isAndroid = /Android/i.test(ua);
  /* 카카오톡·팀즈·네이버·라인·밴드·페북·인스타 안의 브라우저에서는 설치가 되지 않는다 */
  var inApp = /KAKAOTALK|Teams|MSTeams|NAVER|Line\/|band|DaumApps|Instagram|FB_IAB|FBAN|FBAV|everytime/i.test(ua);
  var iosOther = isIOS && /CriOS|FxiOS|EdgiOS|Whale/i.test(ua);

  /* 이미 설치된 앱으로 열렸으면 아무것도 띄우지 않는다 */
  if(standalone) return;
  if(foot) foot.hidden = flag(DONE);

  /* 기기에 맞는 쪽을 먼저 펴 두고, 탭으로 다른 쪽도 볼 수 있게 한다.
     동료 폰 설치를 도와줄 때도 쓰도록. */
  var tab = isIOS ? "ios" : (isAndroid ? "android" : "pc");

  function iosGuide(){
    return (iosOther
        ? '<p class="iwarn">지금 사파리가 아닌 브라우저입니다. <b>사파리</b>로 열면 가장 확실합니다.</p>' : '') +
      '<p><b>아이폰 · 아이패드 (사파리)</b></p>' +
      '<ol>' +
        '<li>화면 아래 <b>공유 버튼</b>(네모에 위쪽 화살표)을 누릅니다. 안 보이면 <b>⋯</b>를 먼저 누릅니다.</li>' +
        '<li>목록을 내려 <b>홈 화면에 추가</b>를 누릅니다.</li>' +
        '<li>오른쪽 위 <b>추가</b>를 누르면 끝입니다.</li>' +
      '</ol>';
  }
  function androidGuide(){
    if(deferred){
      return '<p><b>안드로이드</b></p><ol><li>위의 <b>설치</b> 버튼을 누릅니다.</li>' +
             '<li>뜨는 창에서 <b>설치</b>를 한 번 더 누르면 홈 화면에 아이콘이 생깁니다.</li></ol>';
    }
    return '<p><b>안드로이드 (크롬)</b></p>' +
      '<ol>' +
        '<li>오른쪽 위 <b>메뉴(⋮)</b>를 누릅니다.</li>' +
        '<li><b>앱 설치</b> 또는 <b>홈 화면에 추가</b>를 누릅니다.</li>' +
        '<li><b>설치</b>를 누르면 홈 화면에 아이콘이 생깁니다.</li>' +
      '</ol>' +
      '<p><b>삼성 인터넷</b> — 아래 <b>메뉴(≡)</b> → <b>현재 페이지 추가</b> → <b>홈 화면</b></p>' +
      '<p class="tip">메뉴에 설치 항목이 없으면 페이지를 한 번 새로 고친 뒤 다시 열어 보세요. ' +
      '그래도 없으면 크롬으로 열어 주세요.</p>';
  }
  function pcGuide(){
    return '<p><b>컴퓨터 (크롬 · 엣지)</b></p>' +
      '<ol><li>주소창 오른쪽 끝의 <b>설치 아이콘</b>을 누릅니다.</li>' +
      '<li>없으면 메뉴(⋮ 또는 ⋯) → <b>앱 설치</b>를 누릅니다.</li></ol>';
  }
  function inAppGuide(){
    return '<p class="iwarn"><b>지금 카카오톡·팀즈 같은 앱 안에서 열려 있습니다.</b><br>' +
      '여기서는 설치가 되지 않습니다. 먼저 브라우저로 옮겨 주세요.<br>' +
      '· 카카오톡 안드로이드 — 오른쪽 위 <b>⋮</b> → <b>다른 브라우저로 열기</b><br>' +
      '· 카카오톡 아이폰 — 오른쪽 아래 <b>⋯</b> 또는 공유 → <b>Safari로 열기</b><br>' +
      '· 팀즈·네이버 등 — 메뉴(⋯ ⋮) → <b>브라우저로 열기</b><br>' +
      '옮긴 화면에서 아래 방법대로 설치하면 됩니다.</p>';
  }

  function render(){
    var body = tab === "ios" ? iosGuide() : (tab === "android" ? androidGuide() : pcGuide());
    steps.innerHTML = (inApp ? inAppGuide() : "") +
      '<div class="tabs" role="group" aria-label="설치 안내 플랫폼">' +
        '<button type="button" data-t="ios" aria-pressed="' + (tab==="ios") + '" class="' + (tab==="ios"?"on":"") + '">아이폰</button>' +
        '<button type="button" data-t="android" aria-pressed="' + (tab==="android") + '" class="' + (tab==="android"?"on":"") + '">안드로이드</button>' +
        '<button type="button" data-t="pc" aria-pressed="' + (tab==="pc") + '" class="' + (tab==="pc"?"on":"") + '">컴퓨터</button>' +
      '</div>' + body +
      '<p class="tip">설치 후 기본 화면과 기기에 저장된 콘텐츠를 오프라인으로 이용할 수 있습니다. 처음 보는 그림·소리·웹툰은 인터넷 연결이 필요할 수 있습니다. 설치 항목이 보이지 않으면 이 브라우저에서는 지원하지 않을 수 있습니다. 웹으로 계속 이용할 수 있습니다.</p>';
  }
  steps.addEventListener("click", function(e){
    var t = e.target && e.target.getAttribute && e.target.getAttribute("data-t");
    if(t){ tab = t; render(); var selected = steps.querySelector('[data-t="'+t+'"]'); if(selected) selected.focus(); }
  });

  /* 배너 머리글과 버튼을 지금 상황에 맞춘다 */
  function head(){
    var open = steps.classList.contains("show");
    if(inApp){
      title.textContent = "먼저 크롬·사파리로 열어 주세요";
      sub.textContent = "카카오톡·팀즈 안에서는 설치가 되지 않습니다.";
      go.textContent = open ? "접기" : "방법 보기";
      return;
    }
    title.textContent = "홈 화면에 설치하기";
    if(deferred){
      sub.textContent = "설치를 누른 뒤 브라우저의 안내를 확인하세요.";
      go.textContent = "설치";
    } else {
      sub.textContent = isIOS ? "사파리 공유 버튼으로 홈 화면에 추가합니다."
                              : "홈 화면에서 앱처럼 열 수 있습니다.";
      go.textContent = open ? "접기" : "방법 보기";
    }
  }

function show(withSteps){
  if(withSteps){render();steps.classList.add("show");}
  head();
  if(withSteps){
    bar.setAttribute("role","dialog");bar.setAttribute("aria-modal","true");
    showDialog(bar);go.focus();
  }else bar.classList.add("show");
}

  x.addEventListener("click", function(){
    mark(DISMISS);
    closeInstall();
    steps.classList.remove("show");
  });

  go.addEventListener("click", function(){
    if(deferred && !inApp){
      var prompt = deferred;
      go.disabled = true;
      sub.textContent = "브라우저의 설치 안내를 확인해 주세요.";
      function failed(){
        deferred = null; go.disabled = false;
        render(); steps.classList.add("show"); head();
        sub.textContent = "설치 안내를 열지 못했습니다. 아래 기기별 설치 방법을 확인해 주세요. 웹으로 계속 이용할 수 있습니다.";
      }
      try{
        Promise.all([prompt.prompt(),prompt.userChoice]).then(function(values){
          var c = values[1];
          deferred = null; go.disabled = false;
          if(c && c.outcome === "accepted") closeInstall();
          else {
            if(steps.classList.contains("show")) render(); head();
            sub.textContent = "설치를 취소했습니다. 웹으로 계속 이용할 수 있습니다.";
          }
        },failed);
      }catch(e){ failed(); }
      return;
    }
    if(steps.classList.contains("show")) steps.classList.remove("show");
    else { render(); steps.classList.add("show"); }
    head();
  });

  /* 맨 아래 「홈 화면에 설치하기」 — 배너를 닫은 뒤에도 언제든 다시 연다 */
  if(foot) foot.addEventListener("click", function(){ show(true); });

  window.addEventListener("beforeinstallprompt", function(e){
    e.preventDefault();
    deferred = e;
    if(steps.classList.contains("show")) render();
    head();
    if(!flag(DISMISS) && !flag(DONE)) bar.classList.add("show");
  });

  window.addEventListener("appinstalled", function(){
    mark(DONE);
    deferred = null;
    closeInstall();
    if(foot) foot.hidden = true;
  });

  /* 휴대폰으로 처음 연 사람에게는 한 번 저절로 띄운다. × 로 닫으면 다시 띄우지 않는다. */
  if(!flag(DISMISS) && !flag(DONE) && (isIOS || isAndroid || inApp)){
    setTimeout(function(){
      if(!bar.classList.contains("show")) show(false);
    }, isAndroid && !inApp ? 2500 : 1200);
  }
})();
</script>
